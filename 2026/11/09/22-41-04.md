Subject: Fixing cygwin DLL initialization in AppContainer Process

Greetings! There is an existing issue in the cygwin.dll where initialization of the DLL fails if it is running in an AppContainer process. The failure happens in winsup\cygwin\mm\shared.cc , where the code calls NtCreateDirectoryObject using the global \\BaseNamedObjects<file://\\BaseNamedObjects> namespace. AppContainer processes are not allowed to create directory objects in the global namespaces, and so this fails.

The fix at a high level is simple. AppContainer processes are given their own private namespace to create directory objects within. So the change is simply:
if(AppContainerProcess())
{
```
  appContainerNamespace = GetAppContainerNamespace
  NtCreateDirectoryObject(appContainerNamespace)
```

} else
{
```
 // do as before change
 NtCreateDirectoryObject(\\BaseNamedObjects\...)
```

}

Also added automated coverage using a native launcher and a Cygwin-linked child which verifies its AppContainer token and build-tree runtime, then tests startup and flock locking. Include the socket event-sharing regression test from the branch.

+public:
+  DWORD
+  grant (const std::wstring& path, PSID sid, DWORD access)
+  {

+                                  new_dacl, NULL);

+      LocalFree (new_dacl);

+  restore ()
+  {
+    DWORD first_error = ERROR_SUCCESS;
+    for (size_t i = backups.size (); i > 0; --i)

+                                  DACL_SECURITY_INFORMATION, NULL, NULL,

+         first_error = error;
+       if (error == ERROR_SUCCESS)

+         }

+
+#include <windows.h>
+#include <sys/socket.h>
+#include <sys/wait.h>

+static void
+fail (const char *message)
+{
+  perror (message);
+  exit (1);
+}

+static int
+check_namespace (int fd)
+{
+  char executable[MAX_PATH];
+  char command[MAX_PATH + 32];
+  DWORD length = GetModuleFileNameA (NULL, executable, sizeof executable);
+  if (!length || length >= sizeof executable)
+    return 1;
+  snprintf (command, sizeof command, "\"%s\" --other-namespace", executable);
+
+  /* A native launch avoids inheriting Cygwin's process information. */
+  const char *testing = getenv ("CYGWIN_TESTING");
+  if (!SetEnvironmentVariableA ("CYGWIN_TESTING",
+                              testing && *testing ? NULL : "1"))
+    return 1;
+  STARTUPINFOA startup = { .cb = sizeof startup };
+  PROCESS_INFORMATION process;
+  BOOL created = CreateProcessA (executable, command, NULL, NULL, FALSE, 0,
+                                NULL, NULL, &startup, &process);
+  BOOL restored = SetEnvironmentVariableA ("CYGWIN_TESTING", testing);
+  if (!created)
+    {
+      fputs ("Could not launch the other socket namespace\n", stderr);
+      return 1;
+    }
+  CloseHandle (process.hThread);
+  DWORD wait = WaitForSingleObject (process.hProcess, 20000);
+  DWORD status = 1;
+  if (wait == WAIT_OBJECT_0)
+    GetExitCodeProcess (process.hProcess, &status);
+  else
+    {
+      TerminateProcess (process.hProcess, 1);
+      WaitForSingleObject (process.hProcess, INFINITE);
+    }
+  CloseHandle (process.hProcess);
+  if (!restored || wait != WAIT_OBJECT_0 || status)
+    {
+      fputs ("Other socket namespace failed\n", stderr);
+      return 1;
+    }
+  if (fcntl (fd, F_GETOWN) != getpid ())
+    {
+      fputs ("Other namespace overwrote a live socket's event state\n",
+            stderr);
+      return 1;
+    }
+  return 0;
+}
