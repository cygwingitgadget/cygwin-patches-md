Thanks for the review.

> This sounds like a similar (although distinct) issue to that in
> https://cygwin.com/pipermail/cygwin-patches/2026q4/015440.html
>
> This is not an area I am hugely familiar with, so perhaps you can help
> me out, by also thinking about if there's any interaction between the
> two fixes?


They fix two different exit paths from a signal handler, and they don't
overlap.  7fd5f9c93 makes call_signal_handler() put back the saved incyg
value when the handler returns.  When the handler siglongjmps out
instead, as OpenSSL's armcap probe does, that code is never reached, and
incyg is whatever longjmp stores.  That is the path this patch fixes.

Tested on current main (2158d1470, which includes 7fd5f9c93),
cross-built for aarch64 and run on Windows 11 ARM64:

```
                                          Main                    main + patch
  once2 (7fd5f9c93)    PASS                 PASS
  once3 (new)                  FAIL                  PASS
  winsup total                  267 PASS       268 PASS
                                              18 FAIL               17 FAIL

```

once3 is the only difference.  Without the patch, poll() sleeps the full
5 s because SIGALRM is never delivered.  The other 17 failures are the
same in both runs (15 LTP EFAULT checks and 2 posix_spawn tests that
depend on the local environment).

> I wonder if you could also attach that reproducer, so I can add it to
> our tests somewhere.


This v2 adds it as winsup.api/pthread/once3.c.  It is once2.c with the
handler leaving through siglongjmp instead of returning.

> Yes, please!


The x86_64 patch follows as a separate mail.  x86_64 jmp_buf has no free
8-byte slot (_JBLEN is 32 and setjmp uses all 0x100 bytes), so it stores
incyg in the unused 16-bit Spare field of the Windows _JUMP_BUFFER
layout (offset 0x5e).  With the same reproducer on x86_64 (built on the
3.6.12 sources): FAIL 5/5 without the patch (poll sleeps the full 5 s),
PASS 5/5 with it (EINTR after 1000 ms).

+/* Test signal delivery after a pthread_once initializer siglongjmps out
+   of a signal handler.
+
+   The initializer runs inside Cygwin (incyg == 1).  It does sigsetjmp,
+   raises a signal and the handler siglongjmps back, so the code after
+   the handler in call_signal_handler never runs.  longjmp must then
+   restore the incyg value seen by setjmp; if it clears it instead,
+   pthread_once's return path wraps it to 0xffffffff and later signals
+   are not delivered while the thread is blocked.  This is the pattern
+   of OpenSSL's AArch64 capability probe in a DLL constructor. */
+
+#include "test.h"
+#include <poll.h>
+#include <setjmp.h>
+#include <signal.h>
+#include <time.h>
+
+static sigjmp_buf jb;
+static volatile sig_atomic_t seen;
+
+static void
+jump_handler (int signo)
+{
+  (void) signo;
+  ++seen;
+  siglongjmp (jb, 1);
+}
+
+static void
+handler (int signo)
+{
+  (void) signo;
+  ++seen;
+}
+
+static void
+initialize (void)
+{
+  if (sigsetjmp (jb, 1) == 0)
+    {
+      raise (SIGUSR1);
+      assert (0);      /* jump_handler must not return */
+    }
+}
+
+int
+main (void)
+{
+  struct sigaction action = {0};
+  stack_t stack = {0};
+  stack.ss_sp = malloc (SIGSTKSZ);
+  stack.ss_size = SIGSTKSZ;
+  assert (stack.ss_sp != NULL);
+  assert (sigaltstack (&stack, NULL) == 0);
+
+  assert (sigemptyset (&action.sa_mask) == 0);
+
+  for (int onstack = 0; onstack < 2; ++onstack)
+    {
+      pthread_once_t once = PTHREAD_ONCE_INIT;
+      action.sa_flags = onstack ? SA_ONSTACK : 0;
+      action.sa_handler = jump_handler;
+      assert (sigaction (SIGUSR1, &action, NULL) == 0);
+      action.sa_handler = handler;
+      assert (sigaction (SIGALRM, &action, NULL) == 0);
+      seen = 0;
+      assert (pthread_once (&once, initialize) == 0);
+      assert (seen == 1);
+
+      struct timespec before, after;
+      assert (clock_gettime (CLOCK_MONOTONIC, &before) == 0);
+      alarm (1);
+      assert (poll (NULL, 0, 5000) == -1 && errno == EINTR);
+      assert (clock_gettime (CLOCK_MONOTONIC, &after) == 0);
+      long elapsed_ms = (after.tv_sec - before.tv_sec) * 1000
+                 + (after.tv_nsec - before.tv_nsec) / 1000000;
+      assert (elapsed_ms < 4000);
+      assert (seen == 2);
+    }
+
+  stack.ss_flags = SS_DISABLE;
+  assert (sigaltstack (&stack, NULL) == 0);
+  free (stack.ss_sp);
+  return 0;
+}
